CREATE DATABASE  EV_Charging_Analytics;
USE EV_Charging_Analytics;
 -- ------------------------------------------------------------
--  stations
CREATE TABLE stations (
    station_id INT PRIMARY KEY,
    station_name VARCHAR(100) NOT NULL,
    city VARCHAR(60) NOT NULL,
    address VARCHAR(150),
    opened_date DATE,
    station_status ENUM('Active','Inactive','Maintenance') NOT NULL DEFAULT 'Active'
);
 
--  chargers 
CREATE TABLE chargers (
    charger_id INT PRIMARY KEY,
    station_id INT NOT NULL,
    charger_type ENUM('AC Slow','AC Fast','DC Fast') NOT NULL,
    power_kw DECIMAL(7,2) NOT NULL,
    charger_status ENUM('Available','In Use','Maintenance','Offline') NOT NULL DEFAULT 'Available',
    CONSTRAINT chk_charger_power CHECK (power_kw > 0),
    CONSTRAINT fk_charger_station FOREIGN KEY (station_id)
        REFERENCES stations(station_id)
        ON UPDATE CASCADE ON DELETE RESTRICT
);
 
--  customers
CREATE TABLE customers (
    customer_id INT PRIMARY KEY,
    customer_name VARCHAR(100) NOT NULL,
    email VARCHAR(120) UNIQUE,
    phone VARCHAR(15),
    city VARCHAR(60) NOT NULL,
    registration_date DATE NOT NULL
);
 
--  vehicles (a customer can own many vehicles)
CREATE TABLE vehicles (
    vehicle_id INT PRIMARY KEY,
    customer_id INT NOT NULL,
    make VARCHAR(40) NOT NULL,
    model VARCHAR(40) NOT NULL,
    registration_number VARCHAR(20) NOT NULL UNIQUE,
    battery_capacity_kwh DECIMAL(6,1) NOT NULL,
    vehicle_type ENUM('Electric Car','Electric Scooter','Electric Bike','Electric Bus') NOT NULL,
    CONSTRAINT chk_battery CHECK (battery_capacity_kwh > 0),
    CONSTRAINT fk_vehicle_customer FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id)
        ON UPDATE CASCADE ON DELETE RESTRICT
);
 
-- membership_plans
CREATE TABLE membership_plans (
    plan_id INT PRIMARY KEY,
    plan_name VARCHAR(40) NOT NULL UNIQUE,
    monthly_fee DECIMAL(8,2) NOT NULL,
    discount_percent DECIMAL(5,2) NOT NULL DEFAULT 0,
    duration_days INT NOT NULL,
    CONSTRAINT chk_plan_fee CHECK (monthly_fee >= 0),
    CONSTRAINT chk_plan_discount CHECK (discount_percent BETWEEN 0 AND 100),
    CONSTRAINT chk_plan_days CHECK (duration_days > 0)
);
 
--  customer_memberships (customer <-> plan)
CREATE TABLE customer_memberships (
    membership_id INT PRIMARY KEY,
    customer_id INT NOT NULL,
    plan_id INT NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    membership_status ENUM('Active','Expired','Cancelled') NOT NULL DEFAULT 'Active',
    CONSTRAINT chk_membership_dates CHECK (end_date > start_date),
    CONSTRAINT fk_membership_customer FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_membership_plan FOREIGN KEY (plan_id)
        REFERENCES membership_plans(plan_id)
        ON UPDATE CASCADE ON DELETE RESTRICT
);

-- charging_sessions
CREATE TABLE charging_sessions (
    session_id INT PRIMARY KEY,
    customer_id INT NOT NULL,
    vehicle_id INT NOT NULL,
    charger_id INT NOT NULL,
    start_time DATETIME NOT NULL,
    end_time DATETIME,
    energy_kwh DECIMAL(9,2) NOT NULL DEFAULT 0,
    cost DECIMAL(10,2) NOT NULL DEFAULT 0,
    session_status ENUM('Completed','In Progress','Cancelled') NOT NULL DEFAULT 'Completed',
    CONSTRAINT chk_energy CHECK (energy_kwh >= 0),
    CONSTRAINT chk_cost CHECK (cost >= 0),
    CONSTRAINT chk_session_times CHECK (end_time IS NULL OR end_time >= start_time),
    CONSTRAINT fk_session_customer FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_session_vehicle FOREIGN KEY (vehicle_id)
        REFERENCES vehicles(vehicle_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_session_charger FOREIGN KEY (charger_id)
        REFERENCES chargers(charger_id)
        ON UPDATE CASCADE ON DELETE RESTRICT
);
 
--  payments (at most one payment per session)
CREATE TABLE payments (
    payment_id INT PRIMARY KEY,
    session_id INT NOT NULL UNIQUE,
    payment_method ENUM('UPI','Card','Cash','Wallet','Net Banking') NOT NULL,
    payment_status ENUM('Paid','Pending','Failed','Refunded') NOT NULL DEFAULT 'Pending',
    payment_date DATETIME,
    CONSTRAINT fk_payment_session FOREIGN KEY (session_id)
        REFERENCES charging_sessions(session_id)
        ON UPDATE CASCADE ON DELETE RESTRICT
);
 
--  maintenance( per charger)
CREATE TABLE maintenance (
    log_id INT PRIMARY KEY,
    charger_id INT NOT NULL,
    issue_description VARCHAR(150) NOT NULL,
    reported_date DATE NOT NULL,
    resolved_date DATE,
    technician_name VARCHAR(60),
    maintenance_status ENUM('Open','In Progress','Resolved') NOT NULL DEFAULT 'Open',
    CONSTRAINT chk_maintenance_dates CHECK (resolved_date IS NULL OR resolved_date >= reported_date),
    CONSTRAINT fk_maintenance_charger FOREIGN KEY (charger_id)
        REFERENCES chargers(charger_id)
        ON UPDATE CASCADE ON DELETE RESTRICT
);
 
--  station_reviews (one review per customer per station)
CREATE TABLE station_reviews (
    review_id INT PRIMARY KEY,
    customer_id INT NOT NULL,
    station_id INT NOT NULL,
    rating TINYINT NOT NULL,
    review_comment VARCHAR(200),
    review_date DATE NOT NULL,
    CONSTRAINT chk_rating CHECK (rating BETWEEN 1 AND 5),
    CONSTRAINT uq_review_customer_station UNIQUE (customer_id, station_id),
    CONSTRAINT fk_review_customer FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_review_station FOREIGN KEY (station_id)
        REFERENCES stations(station_id)
        ON UPDATE CASCADE ON DELETE RESTRICT
);
